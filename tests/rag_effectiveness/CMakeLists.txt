cmake_minimum_required(VERSION 3.20)
project(RAGEffectiveness LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${REPO_ROOT}/common" common)
add_subdirectory("${REPO_ROOT}/mcp_client" mcp_client)
add_subdirectory("${REPO_ROOT}/mcp_server" mcp_server)
add_executable(rag_eval_backend backend.cpp)
target_link_libraries(rag_eval_backend PRIVATE mcp_client CURL::libcurl)
target_compile_options(rag_eval_backend PRIVATE -Wall -Wextra -Wpedantic)
